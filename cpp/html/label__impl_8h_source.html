29 template<typename Func>
30 static std::shared_ptr<Label> FindLabel(const std::vector<std::shared_ptr<Label>>& labels, Func&& predicate, bool reverse = false) {
31 for (size_t i = 0; i < labels.size(); ++i) {
32 const size_t index = !reverse ? i : labels.size() - 1 - i;
33 const auto& label = labels[index];
34 if (predicate(label))
35 return label;
36 // Recursive implementation is fine given the shallow depth of labels.
37 const auto& result = FindLabel(label->GetChildren(), predicate, reverse);
38 if (result != nullptr)
39 return result;
40 }
41 return nullptr;
42 }
