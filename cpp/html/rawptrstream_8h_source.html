118 virtual size_t in_avail() const
119 {
120 // See the comment in seek around the restiction that we do not allow read head to
121 // seek beyond the current size.
122 _ASSERTE(m_current_position <= m_size);

124 SafeInt<size_t> readhead(m_current_position);
125 SafeInt<size_t> writeend(m_size);

181 {

217 virtual bool acquire(_Out_ _CharType*& ptr, _Out_ size_t& count)
218 {
219 count = 0;
220 ptr = nullptr;
221
222 if (!this->can_read()) return false;
223
224 count = in_avail();
225
226 if (count > 0)
227 {
228 ptr = (_CharType*)(m_data+m_current_position);
229 return true;
230 }
231 else
232 {
233 ptr = nullptr;
234
235 // Can only be open for read OR write, not both. If there is no data then
236 // we have reached the end of the stream so indicate such with true.
237 return true;

312 virtual pos_type getpos(std::ios_base::openmode mode) const
313 {
314 if ( ((mode & std::ios_base::in) && !this->can_read()) ||
315 ((mode & std::ios_base::out) && !this->can_write()))
316 return static_cast<pos_type>(traits::eof());
317
318 if (mode == std::ios_base::in)
319 return (pos_type)m_current_position;
320 else if (mode == std::ios_base::out)
321 return (pos_type)m_current_position;
322 else
323 return (pos_type)traits::eof();
324 }

333 virtual pos_type seekpos(pos_type position, std::ios_base::openmode mode)
334 {
335 pos_type beg(0);
336 pos_type end(m_size);
337
338 if (position >= beg)
339 {
340 auto pos = static_cast<size_t>(position);
341
342 // Read head
343 if ((mode & std::ios_base::in) && this->can_read())
344 {
345 if (position <= end)
346 {
347 // We do not allow reads to seek beyond the end or before the start position.
348 update_current_position(pos);
349 return static_cast<pos_type>(m_current_position);
350 }
351 }
352
353 // Write head
354 if ((mode & std::ios_base::out) && this->can_write())
355 {
356 // Update write head and satisfy read requests if any
357 update_current_position(pos);
358
359 return static_cast<pos_type>(m_current_position);
360 }
361 }
362
363 return static_cast<pos_type>(traits::eof());
364 }
365
375 virtual pos_type seekoff(off_type offset, std::ios_base::seekdir way, std::ios_base::openmode mode)
376 {
377 pos_type beg = 0;
378 pos_type cur = static_cast<pos_type>(m_current_position);
379 pos_type end = static_cast<pos_type>(m_size);
380
381 switch ( way )
382 {
383 case std::ios_base::beg:
384 return seekpos(beg + offset, mode);
385
386 case std::ios_base::cur:
387 return seekpos(cur + offset, mode);
388
389 case std::ios_base::end:
390 return seekpos(end + offset, mode);
391
392 default:
393 return static_cast<pos_type>(traits::eof());
394 }

439 bool can_satisfy(size_t) const
440 {
441 // We can always satisfy a read, at least partially, unless the
442 // read position is at the very end of the buffer.
443 return (in_avail() > 0);
444 }

450 int_type read_byte(bool advance = true)
451 {
452 _CharType value;
453 auto read_size = this->read(&value, 1, advance);
454 return read_size == 1 ? static_cast<int_type>(value) : traits::eof();
455 }
456
462 size_t read(_Out_writes_ (count) _CharType *ptr, _In_ size_t count, bool advance = true)
463 {
464 if (!can_satisfy(count))
465 return 0;
466
467 SafeInt<size_t> request_size(count);
468 SafeInt<size_t> read_size = request_size.Min(in_avail());
469
470 size_t newPos = m_current_position + read_size;
471
472 auto readBegin = m_data + m_current_position;
473 auto readEnd = m_data + newPos;
474
475#ifdef _WIN32
476#pragma warning(push)
477#pragma warning(disable: 4996) // Disable warning 4996 (checked_array_iterator needs to be replaced)
478 // Avoid warning C4996: Use checked iterators under SECURE_SCL
479 std::copy(readBegin, readEnd, stdext::checked_array_iterator<_CharType *>(ptr, count));
480#pragma warning(pop)
481#else
482 std::copy(readBegin, readEnd, ptr);
483#endif // _WIN32
484
485 if (advance)
486 {
487 update_current_position(newPos);
488 }
489
490 return (size_t) read_size;
491 }
492
496 size_t write(const _CharType *ptr, size_t count)
497 {
498 if (!this->can_write() || (count == 0)) return 0;
499
500 SafeInt<size_t> newSize = SafeInt<size_t>(count) +m_current_position;
501
502 if ( newSize > m_size )
503 throw std::runtime_error("Writing past the end of the buffer");
504
505 // Copy the data
506#ifdef _WIN32
507#pragma warning(push)
508#pragma warning(disable: 4996) // Disable warning 4996 (checked_array_iterator needs to be replaced)
509 // Avoid warning C4996: Use checked iterators under SECURE_SCL
510 std::copy(ptr, ptr + count, stdext::checked_array_iterator<_CharType *>(m_data, m_size, m_current_position));
511#pragma warning(pop)
512#else
513 std::copy(ptr, ptr + count, m_data+m_current_position);
514#endif // _WIN32
515
516 // Update write head and satisfy pending reads if any
517 update_current_position(newSize);
518
519 return count;
520 }
521
525 void update_current_position(size_t newPos)
526 {
527 // The new write head
528 m_current_position = newPos;
529
530 _ASSERTE(m_current_position <= m_size);
531 }
532
533 // The actual memory block
534 _CharType* m_data;
535
536 // The size of the memory block
537 size_t m_size;
538
539 // Read/write head
540 size_t m_current_position;
