60/**
61 * @brief Interface for all the state needed to execute the engine.
62 *
63 * @note Clients should only call the methods to obtain the state that is needed.
64 * Hence, for efficiency, clients may want to implement this interface such that
65 * the corresponding state is computed dynamically instead of computing in advance.
66 */
67class ExecutionState {
68public:
69 /**
70 * @brief Gets the sensitivity label ID that should be applied on the document.
71 *
72 * @return sensitivity label ID to be applied to the content if exists else empty to remove label.
73 */
74 virtual std::shared_ptr<Label> GetNewLabel() const = 0;
75
76 /**
77 * @brief Gets the content description that describes the document.
78 * example for a file: [path\filename]
79 * example for an email: [Subject:Sender]
80 *
81 * @return content description to be applied to the content.
82 *
83 * @note This value is used by auditing as a human-readable description of the content
84 */
85 virtual std::string GetContentIdentifier() const = 0;
86
87 /**
88 * @brief Gets the state of the content while the application is interacting with it.
89 *
90 * @return State of the content data
91 */
92 virtual DataState GetDataState() const {
93 return DataState::USE;
94 };
95
96 /**
97 * @brief Implementation should pass if justification to downgrade an existing label was given.
98 *
99 * @return true if downgrade is justified--along with the justification message--else false
100 * @see mip::JustifyAction
101 */
102 virtual std::pair<bool, std::string> IsDowngradeJustified() const = 0;
103
104 /**
105 * @brief Get the new label's assignment method.
106 *
107 * @return the assignment method STANDARD, PRIVILEGED, AUTO.
108 * @see mip::AssignmentMethod
109 */
110 virtual AssignmentMethod GetNewLabelAssignmentMethod() const = 0;
111
112 /**
113 * @brief Return new label's extended properties
114 *
115 * @return the extended properties applied to the content.
116 */
117 virtual std::vector<std::pair<std::string, std::string>> GetNewLabelExtendedProperties() const {
118 return std::vector<std::pair<std::string, std::string>>();
119 }
120
121 /**
122 * @brief Get the meta-data items from the content.
123 *
124 * @return the metadata applied to the content.
125 * @note Each metadata item is a pair of name and value.
126 */
127 virtual std::vector<std::pair<std::string, std::string>> GetContentMetadata(
128 const std::vector<std::string>& names,
129 const std::vector<std::string>& namePrefixes) const = 0;
130
131 /**
132 * @brief Get the Protection Descriptor
133 *
134 * @return the Protection Descriptor
135 */
136 virtual std::shared_ptr<ProtectionDescriptor> GetProtectionDescriptor() const = 0;
137
138 /**
139 * @brief Gets the content format.
140 *
141 * @return DEFAULT, EMAIL
142 * @see mip::ContentFormat
143 */
144 virtual ContentFormat GetContentFormat() const = 0;
145
146 /**
147 * @brief Gets a masked enum describing all the supported action types.
148 *
149 * @return a masked enum describing all the supported action types.
150 *
151 * @note ActionType::Justify must be supported. When a policy and label change requires justification,
152 * a justification action will always be returned.
153 */
154 virtual ActionType GetSupportedActions() const = 0;
155
156 /**
157 * @brief Return a map of classification results.
158 *
159 * @param classificationIds a list of classification IDs.
160 * @return a list of classification results.
161 * @note return nullptr if no classification cycle executed.
162 */
163 virtual std::shared_ptr<ClassificationResults> GetClassificationResults(
164 const std::vector<std::shared_ptr<ClassificationRequest>>& /*classificationIds*/) const {
165 return nullptr;
166 }
167
168 /**
169 * @brief Return a map of application specific audit key-value pairs.
170 *
171 * @return a list of application specific audit metadata
172 *
173 * @note Registered Key:Value pairs
174 * Sender: Email Id for the sender
175 * Recipients: Represents a JSON array of recipients for an email
176 * LastModifiedBy: Email Id for the user who last modified the content
177 * LastModifiedDate: Date the content was last modified
178 */
179 virtual std::map<std::string, std::string> GetAuditMetadata() const {
180 return std::map<std::string, std::string>();
181 }
182
183 /** @cond DOXYGEN_HIDE */
184 virtual ~ExecutionState() { }
185
186protected:
187 ExecutionState() { }
188 /** @endcond */
189};
