366 static void verify_uri(const web::uri& uri)
367 {
368 // Most of the URI schema validation is taken care by URI class.
369 // We only need to check certain things specific to websockets.
370 if (uri.scheme() != _XPLATSTR("ws") && uri.scheme() != _XPLATSTR("wss"))
371 {
372 throw std::invalid_argument("URI scheme must be 'ws' or 'wss'");
373 }
374
375 if (uri.host().empty())
376 {
377 throw std::invalid_argument("URI must contain a hostname.");
378 }
379
380 // Fragment identifiers are meaningless in the context of WebSocket URIs
381 // and MUST NOT be used on these URIs.
382 if (!uri.fragment().empty())
383 {
384 throw std::invalid_argument("WebSocket URI must not contain fragment identifiers");
385 }
386 }
387
388protected:
389 web::uri m_uri;
390 websocket_client_config m_config;
391};

394class websocket_client_task_impl
395{
396
397public:
398 _ASYNCRTIMP websocket_client_task_impl(websocket_client_config config);
399
400 _ASYNCRTIMP virtual ~websocket_client_task_impl() CPPREST_NOEXCEPT;
401
402 _ASYNCRTIMP pplx::task<websocket_incoming_message> receive();
403
404 _ASYNCRTIMP void close_pending_tasks_with_error(const websocket_exception &exc);
405
406 const std::shared_ptr<websocket_client_callback_impl> & callback_client() const { return m_callback_client; };
407
408private:
409 void set_handler();
410
411 // When a message arrives, if there are tasks waiting for a message, signal the topmost one.
412 // Else enqueue the message in a queue.
413 // m_receive_queue_lock : to guard access to the queue & m_client_closed
414 std::mutex m_receive_queue_lock;
415 // Queue to store incoming messages when there are no tasks waiting for a message
416 std::queue<websocket_incoming_message> m_receive_msg_queue;
417 // Queue to maintain the receive tasks when there are no messages(yet).
418 std::queue<pplx::task_completion_event<websocket_incoming_message>> m_receive_task_queue;
419
420 // Initially set to false, becomes true if a close frame is received from the server or
421 // if the underlying connection is aborted or terminated.
422 bool m_client_closed;
423
424 std::shared_ptr<websocket_client_callback_impl> m_callback_client;
425};

455 pplx::task<void> connect(const web::uri &uri)
456 {
457 m_client->callback_client()->verify_uri(uri);
458 m_client->callback_client()->set_uri(uri);
459 auto client = m_client;
460 return m_client->callback_client()->connect().then([client](pplx::task<void> result)
461 {
462 try
463 {
464 result.get();
465 }
466 catch (const websocket_exception& ex)
467 {
468 client->close_pending_tasks_with_error(ex);
469 throw;
470 }
471 });
472 }
