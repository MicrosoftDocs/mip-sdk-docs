132 // This is an abstraction that is built on top of the scheduler to provide these additional functionalities
133 // - Ability to wait on a work item
134 // - Ability to cancel a work item
135 // - Ability to inline work on invocation of RunAndWait
136 class _TaskCollectionImpl
137 {
138 public:

140 typedef _TaskProcHandle _TaskProcHandle_t;
141
142 _TaskCollectionImpl(scheduler_ptr _PScheduler)
143 : _M_pScheduler(_PScheduler)
144 {
145 }
146
147 void _ScheduleTask(_TaskProcHandle_t* _PTaskHandle, _TaskInliningMode _InliningMode)
148 {
149 if (_InliningMode == _ForceInline)
150 {
151 _TaskProcHandle_t::_RunChoreBridge(_PTaskHandle);
152 }
153 else
154 {
155 _M_pScheduler->schedule(_TaskProcHandle_t::_RunChoreBridge, _PTaskHandle);
156 }
157 }
158
159 void _Cancel()
160 {
161 // No cancellation support
162 }
163
164 void _RunAndWait()
165 {
166 // No inlining support yet
167 _Wait();
168 }
169
170 void _Wait()
171 {
172 _M_Completed.wait();
173 }
174
175 void _Complete()
176 {
177 _M_Completed.set();
178 }

180 scheduler_ptr _GetScheduler() const
181 {
182 return _M_pScheduler;
183 }

185 // Fire and forget
186 static void _RunTask(TaskProc_t _Proc, void * _Parameter, _TaskInliningMode _InliningMode)
187 {
188 if (_InliningMode == _ForceInline)
189 {
190 _Proc(_Parameter);
191 }
192 else
193 {
194 // Schedule the work on the ambient scheduler
195 get_ambient_scheduler()->schedule(_Proc, _Parameter);
196 }
197 }
