29 template<class F, class T = bool>
30 pplx::task<T> _do_while(F func)
31 {
32 pplx::task<T> first = func();
33 return first.then([=](bool guard) -> pplx::task<T> {
34 if (guard)
35 return Concurrency::details::_do_while<F,T>(func);
36 else
37 return first;
38 });
39 }

308 //
309 // Efficient read and write.
310 //
311 // The following routines are intended to be used for more efficient, copy-free, reading and
312 // writing of data from/to the stream. Rather than having the caller provide a buffer into which
313 // data is written or from which it is read, the stream buffer provides a pointer directly to the
314 // internal data blocks that it is using. Since not all stream buffers use internal data structures
315 // to copy data, the functions may not be supported by all. An application that wishes to use this
316 // functionality should therefore first try them and check for failure to support. If there is
317 // such failure, the application should fall back on the copying interfaces (putn / getn)
318 //
319

346 virtual bool acquire(_Out_ _CharType*& ptr, _Out_ size_t& count) = 0;
347
354 virtual void release(_Out_writes_(count) _CharType *ptr, _In_ size_t count) = 0;
355
360 virtual std::exception_ptr exception() const = 0;

402 virtual pplx::task<void> close(std::ios_base::openmode mode = std::ios_base::in | std::ios_base::out)
403 {
404 pplx::task<void> closeOp = pplx::task_from_result();
405
406 if (mode & std::ios_base::in && can_read()) {
407 closeOp = _close_read();
408 }
409
410 // After the flush_internal task completed, "this" object may have been destroyed,
411 // accessing the members is invalid, use shared_from_this to avoid access violation exception.
412 auto this_ptr = std::static_pointer_cast<streambuf_state_manager>(this->shared_from_this());
413
414 if (mode & std::ios_base::out && can_write()) {
415 if (closeOp.is_done())
416 closeOp = closeOp && _close_write().then([this_ptr]{}); // passing down exceptions from closeOp
417 else
418 closeOp = closeOp.then([this_ptr] { return this_ptr->_close_write().then([this_ptr]{}); });
419 }
420
421 return closeOp;
422 }

746 {
747 if (this->exception() == nullptr)

753 // Set exception and eof states for async read
754 template<typename _CharType1>
755 pplx::task<_CharType1> create_exception_checked_task(pplx::task<_CharType1> result, std::function<bool(_CharType1)> eof_test, std::ios_base::openmode mode = std::ios_base::in | std::ios_base::out)
756 {
757 auto thisPointer = this->shared_from_this();
758
759 auto func1 = [=](pplx::task<_CharType1> t1) -> pplx::task<_CharType1> {
760 try {
761 thisPointer->m_stream_read_eof = eof_test(t1.get());
762 } catch (...) {
763 thisPointer->close(mode, std::current_exception()).get();
764 return pplx::task_from_exception<_CharType1>(thisPointer->exception(), pplx::task_options());
765 }
766 if (thisPointer->m_stream_read_eof && !(thisPointer->exception() == nullptr))
767 return pplx::task_from_exception<_CharType1>(thisPointer->exception(), pplx::task_options());
768 return t1;
769 };
770
771 if ( result.is_done() )
772 {
773 // If the data is already available, we should avoid scheduling a continuation, so we do it inline.
774 return func1(result);
775 }
776 else
777 {
778 return result.then(func1);
779 }
780 }
