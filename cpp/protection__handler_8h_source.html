129 /**

137 ConsumptionSettings(const std::shared_ptr<PublishingLicenseInfo>& licenseInfo)

142 * @brief Get the publishing license associated with the protected content

148 }
149
150 /**
151 * @brief Gets whether or not ProtectionHandler creation allows for online HTTP operations
152 *

156 * previously decrypted and its unexpired license is cached. A mip::NetworkError will be thrown if cached

159 bool GetIsOfflineOnly() const { return mIsOfflineOnly; }
160

167 * previously decrypted and its unexpired license is cached. A mip::NetworkError will be thrown if cached

171
172 /**
