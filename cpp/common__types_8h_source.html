556class CaseInsensitiveComparator {
557public:
558 bool operator() (const std::string& lhs, const std::string& rhs) const {
559 if (lhs.size() != rhs.size())
560 return lhs.size() < rhs.size();
561
562#ifdef _WIN32
563 return _strnicmp(lhs.c_str(), rhs.c_str(), lhs.size()) < 0;
564#else
565 return strncasecmp(lhs.c_str(), rhs.c_str(), lhs.size()) < 0;
566#endif // WIN32
567 }
