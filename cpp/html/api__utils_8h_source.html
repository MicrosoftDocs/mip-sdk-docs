111 if (!error)

114 try {

143// Wrap a method with error handling and telemetry
144// Throws only mip::Error objects
145// This function returns dynamic type according to the return type of template parameter Func.
146// See http://en.cppreference.com/w/cpp/language/decltype.
147template<typename Func>
148auto WrapApiImpl(
149 Event& event,
150 const std::shared_ptr<TelemetryContext>& apiTelemetryContext,
151 const Func& method,
152 const std::string& file,
153 int32_t line,
154 const std::string& func) -> decltype(method(nullptr)) {
155 try {
156 return method(apiTelemetryContext);
157 } catch (...) {
158 auto error = std::current_exception();
159 //ApiErrorHandler will always throw
160 ApiErrorHandler(
161 error,
162 event,
163 *apiTelemetryContext,
164 file,
165 line,
166 func);
167 throw; //fix for not all control paths return a value error
168 }

188template<typename SuccessFunc, typename ResultType>
189void TryExecuteSuccessCallback(
190 const SuccessFunc& successMethod,
191 const ResultType& result,
192 bool executesOnNewThread,
193 const std::shared_ptr<TaskDispatcherDelegate>& taskDispatcher,
194 const std::string& eventName,
195 const std::shared_ptr<ApiContext>& apiContext) {
196 auto fn = [successMethod, result, eventName]() {
197 // Log success/failure of callback method
198 try {
199 MIPLOG(INFO) << api_helpers::GetStartApiSuccessCallbackMessage(eventName);
200 successMethod(result);
201 MIPLOG(INFO) << api_helpers::GetEndApiSuccessCallbackMessage(eventName);
202 } catch (...) {
203 MIPLOG(WARNING) << "Callback failed for API call: [ " + eventName + " ] " << api_helpers::What(std::current_exception());
204 throw;
205 }
206 };
207
208 if (executesOnNewThread)
209 api_helpers::RunOnIndependentThread(GetNextTaskId("ApiSuccess"), fn, taskDispatcher, apiContext);
210 else
211 fn();
212}

214template<typename FailureFunc>
215void TryExecuteFailureCallback(
216 const FailureFunc& failureMethod,
217 const std::exception_ptr& error,
218 bool executesOnNewThread,
219 const std::shared_ptr<TaskDispatcherDelegate>& taskDispatcher,
220 const std::string& eventName,
221 const std::shared_ptr<ApiContext>& apiContext) {
222 // Wrap current exception in mip error just in case one slipped through
223 std::exception_ptr mipError = nullptr;
224 try {
225 std::rethrow_exception(error);
226 } catch (const Error&) { // Exception should always be mip::Error
227 mipError = std::current_exception();
228 } catch (...) {
229 MIPLOG(MIPERROR) << api_helpers::GetUnrecognizedExceptionMessage(eventName);
230 mipError = std::make_exception_ptr(InternalError(api_helpers::What(std::current_exception())));
231 }
232
233 auto fn = [failureMethod, mipError, eventName]() {
234 // Log success/failure of callback method
235 try {
236 MIPLOG(WARNING) << api_helpers::GetStartApiFailureCallbackMessage(eventName);
237 failureMethod(mipError);
238 MIPLOG(WARNING) << api_helpers::GetEndApiFailureCallbackMessage(eventName);
239 } catch (...) {
240 MIPLOG(WARNING) << "Callback failed for API call: [ " + eventName + " ] " << api_helpers::What(std::current_exception());
241 throw;
242 }
243 };
244
245 if (executesOnNewThread)
246 api_helpers::RunOnIndependentThread(GetNextTaskId("ApiFailure"), fn, taskDispatcher, apiContext);
247 else
248 fn();
249}

283template<typename Func, typename SuccessFunc, typename FailureFunc>
284void WrapObserverApiImpl(
285 const std::string& eventName,
286 const std::shared_ptr<TelemetryContext>& parentTelemetryContext,
287 const std::shared_ptr<MipContext>& mipContext,
288 const Func& method,
289 const SuccessFunc& successMethod,
290 const FailureFunc& failureMethod,
291 const std::shared_ptr<TaskDispatcherDelegate>& taskDispatcher,
292 const std::shared_ptr<ApiContext>& apiContext,
293 const std::string& file,
294 int32_t line,
295 const std::string& func,
296 const std::function<void(std::function<void()>&&)>& taskFunc) {
297 // mipContext is validated by the caller
298 ApiContextLifetime contextLifeTime(apiContext);
299 auto event = CreateTelemetryEvent(
300 eventName,
301 EventLevel::Basic);
302 auto eventContext = parentTelemetryContext ? parentTelemetryContext->GetEventContext() : api_helpers::GetDefaultEventContext(*mipContext);
303 auto apiTelemetryContext = CreateTelemetryContext("" /*correlationId*/, "API:" + eventName, eventContext, parentTelemetryContext);
304 std::string scenarioId = GenerateGuid();
305 MIPLOGEX(INFO, file, line, func) << api_helpers::GetStartApiMessage(eventName + " scenarioId=" + scenarioId);
306
307 std::function<decltype(method(nullptr))(const std::shared_ptr<TelemetryContext>&)> methodFunc(method);
308 auto wrapperAsyncFn = [
309 methodFunc,
310 successMethod,
311 failureMethod,
312 taskDispatcher,
313 eventName,
314 event,
315 apiTelemetryContext,
316 mipContext,
317 scenarioId,
318 file,
319 line,
320 func,
321 apiContext] () mutable {
322 ApiContextLifetime contextLifeTime(apiContext);
323 MIPLOGEX(INFO, file, line, func) << api_helpers::GetStartApiTaskMessage(eventName + " scenarioId=" + scenarioId);
324 decltype(methodFunc(nullptr)) result;
325 std::exception_ptr currentException = nullptr;
326
327 auto telemetryManager = mipContext->GetTelemetryManager();
328 try {
329 auto autoTelemetryEvent = CreateAutoTelemetryEvent(event, apiTelemetryContext, telemetryManager);
330 result = api_helpers::WrapApiImpl(autoTelemetryEvent->Get(), apiTelemetryContext, methodFunc, file, line, func);
331 } catch (...) {
332 currentException = std::current_exception();
333 }
334 methodFunc = nullptr; // Release resources in callback function
335
336 if (currentException)
337 TryExecuteFailureCallback(failureMethod, currentException, true /*executesOnNewThread*/, taskDispatcher, eventName, apiContext);
338 else
339 TryExecuteSuccessCallback(successMethod, result, true /*executesOnNewThread*/, taskDispatcher, eventName, apiContext);
340
341 MIPLOGEX(INFO, file, line, func) << api_helpers::GetEndApiTaskMessage(eventName);
342 telemetryManager->ReleaseThreadResources(); // Clean up std::thread/EventLoop thread
343 };
344
345 methodFunc = nullptr; // Release resources in callback function
346 taskFunc(std::move(wrapperAsyncFn));
347 MIPLOGEX(INFO, file, line, func) << api_helpers::GetEndApiMessage(eventName);
348}

350template<typename Func, typename SuccessFunc, typename FailureFunc>
351std::shared_ptr<AsyncControl> WrapAsyncApiObserverImpl(
352 const std::string& eventName,
353 const std::shared_ptr<TelemetryContext>& parentTelemetryContext,
354 const std::shared_ptr<MipContext>& mipContext,
355 const std::string& taskId,
356 const Func& method,
357 const SuccessFunc& successMethod,
358 const FailureFunc& failureMethod,
359 bool executesOnNewThread,
360 const std::shared_ptr<TaskDispatcherDelegate>& taskDispatcher,
361 const std::shared_ptr<void>& loggerContext,
362 const std::string& file,
363 int32_t line,
364 const std::string& func) {

367 }

398 api_helpers::GetEndApiMessage(eventName),
399 func,
400 file,

417 auto wrappedFailureMethod =
418 [&failureMethod, &eventName, &autoTelemetryEvent, &apiTelemetryContext, &file, &line, &func, &apiContext](const std::exception_ptr& ex) {

434template<typename SuccessReturnType, typename GetAsyncFunc, typename SuccessFunc, typename FailureFunc>
435std::shared_ptr<AsyncControl> WrapAsyncApiCallbackImpl(
436 const std::string& eventName,
437 const std::shared_ptr<TelemetryContext>& parentTelemetryContext,
438 const MipContext& mipContext,
439 const GetAsyncFunc& getAsyncMethod,
440 const SuccessFunc& successMethod,
441 const FailureFunc& failureMethod,
442 const std::shared_ptr<TaskDispatcherDelegate>& taskDispatcher,
443 const std::shared_ptr<void>& loggerContext,
444 const std::string& file,
445 int32_t line,
446 const std::string& func) {
447 auto apiContext = ApiContext::Create(loggerContext, mipContext.GetMipConfiguration());
448 ApiContextLifetime contextLifeTime(apiContext);
449 auto eventContext = parentTelemetryContext ? parentTelemetryContext->GetEventContext() : api_helpers::GetDefaultEventContext(mipContext);
450 auto apiTelemetryContext = CreateTelemetryContext("" /*correlationId*/, "API:" + eventName, eventContext, parentTelemetryContext);
451 auto event = CreateTelemetryEvent(
452 eventName,
453 EventLevel::Basic);
454
455 std::string scenarioId = GenerateGuid();
456 MIPLOGEX(INFO, file, line, func) << api_helpers::GetStartApiMessage(eventName + " scenarioId=" + scenarioId);
457

460 auto wrappedSuccessMethod =
461 [successMethod, eventName, event, apiTelemetryContext, telemetryManager, func, file, line, taskDispatcher, apiContext] (const SuccessReturnType& result) {
462 auto autoTelemetryEvent = CreateAutoTelemetryEvent(event, apiTelemetryContext, telemetryManager);
463 TryExecuteSuccessCallback(successMethod, result, true /*executesOnNewThread*/, taskDispatcher, eventName, apiContext);
464 MIPLOGEX(INFO, file, line, func) << api_helpers::GetEndApiTaskMessage(eventName);
465 };
466
467 auto wrappedFailureMethod =
468 [failureMethod, eventName, event, apiTelemetryContext, telemetryManager, func, file, line, taskDispatcher, apiContext](const std::exception_ptr& error) {

472 api_helpers::ApiErrorHandler(
473 error,

489 MIPLOGEX(INFO, file, line, func) << api_helpers::GetEndApiTaskMessage(eventName);
490 };
491
492 // Ensure that any failure in this execution arrives at the failureFn instead of throwing to event loop
493 auto wrappedAsyncMethod = [
494 getAsyncMethod,
495 eventName,
496 func,
497 file,
498 line,
499 scenarioId] (

506 } catch (...) {

508 }
509 };
510

512 taskDispatcher->DispatchTask(
513 taskId,
514 [wrappedAsyncMethod, wrappedSuccessMethod, wrappedFailureMethod, eventName, telemetryManager, func, file, line, apiTelemetryContext, apiContext]() {
515 ApiContextLifetime contextLifeTime(apiContext);
516 MIPLOGEX(INFO, file, line, func) << api_helpers::GetStartApiTaskMessage(eventName);
517 wrappedAsyncMethod(wrappedSuccessMethod, wrappedFailureMethod, apiTelemetryContext);
518 telemetryManager->ReleaseThreadResources(); // Clean up ThreadLoop thread
519 },
520 loggerContext);
521
522 MIPLOGEX(INFO, file, line, func) << api_helpers::GetEndApiMessage(eventName);
523 return std::make_shared<AsyncControlImpl>(taskId, taskDispatcher, loggerContext);
524}
