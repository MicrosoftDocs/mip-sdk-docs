182 virtual bool acquire(_Out_ _CharType*& ptr, _Out_ size_t& count)
183 {
184 ptr = nullptr;
185 count = 0;

189 count = in_avail();

272 virtual pos_type getpos(std::ios_base::openmode mode) const

288 virtual pos_type seekpos(pos_type position, std::ios_base::openmode mode)
289 {
290 pos_type beg(0);
291
292 // In order to support relative seeking from the end position we need to fix an end position.
293 // Technically, there is no end for the stream buffer as new writes would just expand the buffer.
294 // For now, we assume that the current write_end is the end of the buffer. We use this artificial
295 // end to restrict the read head from seeking beyond what is available.
296
297 pos_type end(m_data.size());
298
299 if (position >= beg)
300 {
301 auto pos = static_cast<size_t>(position);
302
303 // Read head
304 if ((mode & std::ios_base::in) && this->can_read())
305 {
306 if (position <= end)
307 {
308 // We do not allow reads to seek beyond the end or before the start position.
309 update_current_position(pos);
310 return static_cast<pos_type>(m_current_position);
311 }
312 }
313
314 // Write head
315 if ((mode & std::ios_base::out) && this->can_write())
316 {
317 // Allocate space
318 resize_for_write(pos);
319
320 // Nothing to really copy
321
322 // Update write head and satisfy read requests if any
323 update_current_position(pos);
324
325 return static_cast<pos_type>(m_current_position);
326 }
327 }
328
329 return static_cast<pos_type>(traits::eof());
330 }
331
341 virtual pos_type seekoff(off_type offset, std::ios_base::seekdir way, std::ios_base::openmode mode)
342 {
343 pos_type beg = 0;
344 pos_type cur = static_cast<pos_type>(m_current_position);
345 pos_type end = static_cast<pos_type>(m_data.size());
346
347 switch ( way )
348 {
349 case std::ios_base::beg:
350 return seekpos(beg + offset, mode);
351
352 case std::ios_base::cur:
353 return seekpos(cur + offset, mode);
354
355 case std::ios_base::end:
356 return seekpos(end + offset, mode);
357
358 default:
359 return static_cast<pos_type>(traits::eof());
360 }
361 }

387 static void validate_mode(std::ios_base::openmode mode)
388 {
389 // Disallow simultaneous use of the stream buffer for writing and reading.
390 if ((mode & std::ios_base::in) && (mode & std::ios_base::out))
391 throw std::invalid_argument("this combination of modes on container stream not supported");
392 }

397 bool can_satisfy(size_t)
398 {
399 // We can always satisfy a read, at least partially, unless the
400 // read position is at the very end of the buffer.
401 return (in_avail() > 0);
402 }
403
408 int_type read_byte(bool advance = true)
409 {
410 _CharType value;
411 auto read_size = this->read(&value, 1, advance);
412 return read_size == 1 ? static_cast<int_type>(value) : traits::eof();
413 }
414
420 size_t read(_Out_writes_ (count) _CharType *ptr, _In_ size_t count, bool advance = true)
421 {
422 if (!can_satisfy(count))
423 return 0;
424
425 SafeInt<size_t> request_size(count);
426 SafeInt<size_t> read_size = request_size.Min(in_avail());
427
428 size_t newPos = m_current_position + read_size;
429
430 auto readBegin = std::begin(m_data) + m_current_position;
431 auto readEnd = std::begin(m_data) + newPos;
432
433#ifdef _WIN32
434#pragma warning(push)
435#pragma warning(disable: 4996) // Disable warning 4996 (checked_array_iterator needs to be replaced with span)
436 // Avoid warning C4996: Use checked iterators under SECURE_SCL
437 std::copy(readBegin, readEnd, stdext::checked_array_iterator<_CharType *>(ptr, count));
438#pragma warning(pop)
439#else
440 std::copy(readBegin, readEnd, ptr);
441#endif // _WIN32
442
443 if (advance)
444 {
445 update_current_position(newPos);
446 }
447
448 return (size_t) read_size;
449 }
450
454 size_t write(const _CharType *ptr, size_t count)
455 {
456 if (!this->can_write() || (count == 0)) return 0;
457
458 auto newSize = m_current_position + count;
459
460 // Allocate space
461 resize_for_write(newSize);
462
463 // Copy the data
464 std::copy(ptr, ptr + count, std::begin(m_data) + m_current_position);
465
466 // Update write head and satisfy pending reads if any
467 update_current_position(newSize);
468
469 return count;
470 }
471
475 void resize_for_write(size_t newPos)
476 {
477 // Resize the container if required
478 if (newPos > m_data.size())
479 {
480 m_data.resize(newPos);
481 }
482 }
483
487 void update_current_position(size_t newPos)
488 {
489 // The new write head
490 m_current_position = newPos;
491 _ASSERTE(m_current_position <= m_data.size());
492 }
493
494 // The actual data store
495 _CollectionType m_data;
496
497 // Read/write head
498 size_t m_current_position;
