226 int_type overflow(int_type ch)
227 {
228 try
229 {
230 return m_buffer.putc(CharType(ch)).get();
231 }
232 catch(...)

246 }
247 catch(...)

256 int_type uflow()

271 std::streamsize xsgetn(_Out_writes_ (count) CharType* ptr, _In_ std::streamsize count)
272 {
273 size_t cnt = size_t(count);
274 size_t read_so_far = 0;
275
276 try
277 {
278 while (read_so_far < cnt)
279 {
280 size_t rd = m_buffer.getn(ptr+read_so_far, cnt-read_so_far).get();
281 read_so_far += rd;
282 if ( rd == 0 )
283 break;
284 }
285 return read_so_far;
286 }
287 catch(...)
288 {
289 return 0;
290 }
291 }
292
296 std::streamsize xsputn(const CharType* ptr, std::streamsize count)
297 {
298 try
299 {
300 return m_buffer.putn_nocopy(ptr, static_cast<size_t>(count)).get();
301 }
302 catch(...)
303 {
304 return 0;
305 }
306 }
307
311 int sync() // must be int as per std::basic_streambuf
312 {
313 try
314 {
315 m_buffer.sync().wait();
316 }
317 catch(...)
318 {
319 }
320 return 0;
321 }

326 pos_type seekoff(off_type offset,
327 std::ios_base::seekdir dir,
328 std::ios_base::openmode mode = std::ios_base::in | std::ios_base::out)
329 {
330 try
331 {
332 if ( dir == std::ios_base::cur && offset == 0) // Special case for getting the current position.
333 return m_buffer.getpos(mode);
334 return m_buffer.seekoff(offset,dir,mode);
335 }
336 catch(...)
337 {
338 return (pos_type(-1));
339 }
340 }
