160 class recursive_lock_impl
161 {
162 public:
163
164 recursive_lock_impl()
165 : _M_owner(-1), _M_recursionCount(0)
166 {
167 }
168
169 ~recursive_lock_impl()
170 {
171 _ASSERTE(_M_owner == -1);
172 _ASSERTE(_M_recursionCount == 0);
173 }
174
175 void lock()
176 {
177 auto id = ::pplx::details::platform::GetCurrentThreadId();
178
179 if ( _M_owner == id )
180 {
181 _M_recursionCount++;
182 }
183 else
184 {
185 _M_cs.lock();
186 _M_owner = id;
187 _M_recursionCount = 1;
188 }
189 }
190
191 void unlock()
192 {
193 _ASSERTE(_M_owner == ::pplx::details::platform::GetCurrentThreadId());
194 _ASSERTE(_M_recursionCount >= 1);
195
196 _M_recursionCount--;
197
198 if ( _M_recursionCount == 0 )
199 {
200 _M_owner = -1;
201 _M_cs.unlock();
202 }
203 }
204
205 private:
206 pplx::details::critical_section_impl _M_cs;
207 long _M_recursionCount;
208 volatile long _M_owner;
209 };
