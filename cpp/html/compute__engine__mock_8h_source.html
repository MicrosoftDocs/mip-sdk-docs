114protected:
115 std::vector<std::shared_ptr<Action>> RunComputeActions(
116 const DocumentState&,
117 const ApplicationActionState& actionState,
118 const std::shared_ptr<Label>& label,
119 ComputeEngineContextBase& context) override {
120 UNUSED_PARAMETER(context);
121 auto result = std::vector<std::shared_ptr<Action>>();
122 if (label) {
123 auto it = mActions.find(label->GetId());
124 auto supportedActionTypes = actionState.GetSupportedActions();
125 if (it != mActions.end()) {
126 for (const auto& action : it->second) {
127 if (static_cast<unsigned int>(supportedActionTypes & action->GetType())) {
128 result.push_back(action);
129 }
130 }
131 }
132 }
133 return result;
134 }

136private:
137 std::map<std::string, std::vector<std::shared_ptr<Action>>> mActions;
