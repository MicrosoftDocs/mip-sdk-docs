49 bool AcquireOAuth2Token(
50 const mip::Identity& /* identity */,
51 const OAuth2Challenge& challenge,
52 const std::shared_ptr<void>& /* context */,
53 OAuth2Token& token) {
54 if (challenge.GetResource() == GetSccResource()) {
55 if (!mSccToken.empty()) {
56 token.SetAccessToken(mSccToken);
57 return true;
58 }
59 } else {
60 if (!mProtectionToken.empty()) {
61 token.SetAccessToken(mProtectionToken);
62 return true;
63 }
64 }
65 return false;
66 }
67
68private:
69 std::string mSccToken;
70 std::string mProtectionToken;
