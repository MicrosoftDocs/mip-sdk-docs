107// "CPDF_Document" is responsible for the release of the indirect object which is obtained from
108// "CPDF_Document::GetIndirectobject"，In order to avoid increasing memory usage due to too many objects loaded, we
109// can release indirect object by "CPDF_ Document:: ReleaseIndirectobject". But some objects need to exist between the whole
110// "CPDF_Document"'s existence, such as info and root dictionary. We use "AutoIndirectObj" to obtain indirect object. The flag
111// "mIsTemp" is set in it to judge whether to release the object actively after using. If the object with the specified
112// object number has loaded before AutoIndirectObj construct, "mIsTemp" will be set to false, otherwise "mIsTemp" will be
113// set to true.
114class AutoIndirectObj {
115public:
116 static AutoIndirectObj Load(CPDF_Document *doc, FX_UINT32 objNum);
117
118 CPDF_Object *operator->() { return mObject; }
119
120 operator CPDF_Object *() { return mObject; }
121
122 ~AutoIndirectObj();
123
124 CPDF_Object *Get() const { return mObject; }
125
126 AutoIndirectObj(const AutoIndirectObj &) = delete;
127
128 AutoIndirectObj &operator=(const AutoIndirectObj &) = delete;
129
130 AutoIndirectObj(AutoIndirectObj &&other);
131
132 AutoIndirectObj &operator=(AutoIndirectObj &&other);
133
134private:
135 AutoIndirectObj(CPDF_Document *doc, CPDF_Object *obj, bool temp) : mDoc(doc), mObject(obj), mIsTemp(temp) {}

138 CPDF_Object *mObject;
