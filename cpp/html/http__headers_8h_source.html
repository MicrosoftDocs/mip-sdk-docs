213 {
214 auto iter = m_headers.find(name);
215 if (iter != m_headers.end())
216 {
217 // Check to see if doesn't have a value.

296 bool bind_impl(const key_type &text, _t &ref) const
297 {
298 utility::istringstream_t iss(text);
299 iss.imbue(std::locale::classic());
300 iss >> ref;

302 {
303 return false;

305
