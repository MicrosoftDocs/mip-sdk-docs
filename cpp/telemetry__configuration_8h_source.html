44/**
45 * @brief Custom telemetry settings (not commonly used)
46 */
47struct TelemetryConfiguration {
48 std::string hostNameOverride; /**< Host telemetry instance name. If not set, MIP will act as its own host. */
49 std::string libraryNameOverride; /**< Alternate telemetry library (DLL) filename. */
50 std::shared_ptr<HttpDelegate> httpDelegateOverride; /**< If set, HTTP handling will be managed by this instance*/
51 std::shared_ptr<TaskDispatcherDelegate> taskDispatcherDelegateOverride; /**< If set, async task handling will be managed by this instance,
52 taskDispatcherDelegateOverides should not be shared as they can hold telemetry objects, and prevent their release until taskDispatcher is freed.*/
53 bool isNetworkDetectionEnabled {true}; /**< If set, telemetry component will ping network status on background thread*/
54 bool isLocalCachingEnabled {true}; /**< If set, telemetry component will use on-disk caching */
55 bool isTraceLoggingEnabled {false}; /**< If set, telemetry component will write warning/error logs to disk */
56 bool isTelemetryOptedOut {false}; /**< If set, only necessary service data telemetry will be sent */
57 bool isFastShutdownEnabled {false}; /**< If set, No events will be uploaded on shutdown, Audit events will be uploaded immediately upon logging */
58 std::map<std::string, std::string> customSettings; /**< Custom telemetry settings >**/
59 std::map<std::string, std::vector<std::string>> maskedProperties; /**< Telemetry events/properties which should be masked */
60};

62} // namespace mip
