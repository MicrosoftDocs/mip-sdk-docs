39 // This operation queue is NOT thread safe
40 class async_operation_queue
41 {
42 pplx::task<void> m_lastOperation;
43 public:
44 async_operation_queue()
45 {
46 m_lastOperation = pplx::task_from_result();
47 }
48
49 // It only accepts functors that take no argument and return pplx::task<T>
50 // This function may execute op inline, thus it could throw immediately
51 template <typename Func>
52 auto enqueue_operation(Func &&op) -> decltype(op())
53 {
54 decltype(op()) res; // res is task<T> , which always has default constructor
55 if (m_lastOperation.is_done())
56 {
57 res = op(); // Exceptions are expected to be thrown directly without catching
58 if (res.is_done())
59 return res;
60 }
61 else
62 {
63 res = m_lastOperation.then([=] {
64 return op(); // It will cause task unwrapping
65 });
66 }
67 m_lastOperation = res.then([&] (decltype(op())) {
68 // This empty task is necessary for keeping the rest of the operations on the list running
69 // even when the previous operation gets error.
70 // Don't observe exception here.
71 });
72 return res;
73 }
74
75 void wait() const
76 {
77 m_lastOperation.wait();
78 }
79 };

173 size_t _in_avail_unprot() const
174 {
175 if ( !this->is_open() ) return 0;
176
177 if ( m_info->m_buffer == nullptr || m_info->m_buffill == 0 ) return 0;
178 if ( m_info->m_bufoff > m_info->m_rdpos || (m_info->m_bufoff+m_info->m_buffill) < m_info->m_rdpos ) return 0;
179
180 SafeInt<size_t> rdpos(m_info->m_rdpos);
181 SafeInt<size_t> buffill(m_info->m_buffill);
182 SafeInt<size_t> bufpos = rdpos - m_info->m_bufoff;
183
184 return buffill - bufpos;
185 }

195 static pplx::task<void> _close_file(_In_ _file_info * fileInfo)
196 {
197 pplx::task_completion_event<void> result_tce;
198 auto callback = new _filestream_callback_close(result_tce);
199
200 if ( !_close_fsb_nolock(&fileInfo, callback) )

208 // Workaround GCC compiler bug https://gcc.gnu.org/bugzilla/show_bug.cgi?id=58972
209 void _invoke_parent_close_read()
210 {
211 streambuf_state_manager<_CharType>::_close_read();
212 }
213
214 pplx::task<void> _close_read()
215 {
216 return m_readOps.enqueue_operation([this]
217 {
218 _invoke_parent_close_read();
219
220 if (this->can_write())
221 {
222 return pplx::task_from_result();
223 }
224 else
225 {
226 // Neither heads are open. Close the underlying device
227 // to indicate that we are no longer open
228 auto fileInfo = _close_stream();
229
230 return _close_file(fileInfo);
231 }
232 });
233 }
234
235 pplx::task<void> _close_write()
236 {
237 streambuf_state_manager<_CharType>::_close_write();
238 if (this->can_read())
239 {
240 // Read head is still open. Just flush the write data
241 return flush_internal();
242 }
243 else
244 {
245 // Neither heads are open. Close the underlying device
246
247 // We need to flush all writes if the file was opened for writing.
248 return flush_internal().then([=](pplx::task<void> flushTask) -> pplx::task<void>
249 {
250 // swallow exception from flush
251 try
252 {
253 flushTask.wait();
254 }
255 catch(...)
256 {
257 }
258
259 // indicate that we are no longer open
260 auto fileInfo = this->_close_stream();
261
262 return this->_close_file(fileInfo);
263 });
264 }

272 virtual pplx::task<int_type> _putc(_CharType ch)
273 {
274 auto result_tce = pplx::task_completion_event<size_t>();
275 auto callback = new _filestream_callback_write<size_t>(m_info, result_tce);
276
277 // Potentially we should consider deprecating this API, it is TERRIBLY inefficient.
278 std::shared_ptr<_CharType> sharedCh;
279 try
280 {
281 sharedCh = std::make_shared<_CharType>(ch);
282 } catch (const std::bad_alloc &)
283 {
284 delete callback;
285 throw;
286 }
287
288 size_t written = _putn_fsb(m_info, callback, sharedCh.get(), 1, sizeof(_CharType));
289 if (written == sizeof(_CharType))
290 {
291 delete callback;
292 return pplx::task_from_result<int_type>(ch);
293 }
294
295 return pplx::create_task(result_tce).then([sharedCh](size_t)
296 {
297 return static_cast<int_type>(*sharedCh);
298 });
299 }

393 virtual pplx::task<int_type> _bumpc()
394 {
395 return m_readOps.enqueue_operation([this]()-> pplx::task<int_type> {
396 if ( _in_avail_unprot() > 0 )
397 {
398 pplx::extensibility::scoped_recursive_lock_t lck(m_info->m_lock);
399
400 // Check again once the lock is held.
401
402 if ( _in_avail_unprot() > 0 )
403 {
404 auto bufoff = m_info->m_rdpos - m_info->m_bufoff;
405 _CharType ch = m_info->m_buffer[bufoff*sizeof(_CharType)];
406 m_info->m_rdpos += 1;
407 return pplx::task_from_result<int_type>(ch);
408 }
409 }
410
411 auto result_tce = pplx::task_completion_event<int_type>();
412 auto callback = new _filestream_callback_bumpc(m_info, result_tce);
413
414 size_t ch = _getn_fsb(m_info, callback, &callback->m_ch, 1, sizeof(_CharType));
415
416 if ( ch == sizeof(_CharType) )
417 {
418 pplx::extensibility::scoped_recursive_lock_t lck(m_info->m_lock);
419 m_info->m_rdpos += 1;
420 _CharType ch1 = (_CharType)callback->m_ch;
421 delete callback;
422 return pplx::task_from_result<int_type>(ch1);
423 }
424 return pplx::create_task(result_tce);
425 });

548 virtual pplx::task<size_t> _getn(_Out_writes_ (count) _CharType *ptr, _In_ size_t count)
549 {
550 return m_readOps.enqueue_operation([=] ()-> pplx::task<size_t>{
551 if ( m_info->m_atend || count == 0 )
552 return pplx::task_from_result<size_t>(0);
553
554 if ( _in_avail_unprot() >= count )
555 {
556 pplx::extensibility::scoped_recursive_lock_t lck(m_info->m_lock);
557
558 // Check again once the lock is held.
559
560 if ( _in_avail_unprot() >= count )
561 {
562 auto bufoff = m_info->m_rdpos - m_info->m_bufoff;
563 std::memcpy((void *)ptr, this->m_info->m_buffer+bufoff*sizeof(_CharType), count*sizeof(_CharType));
564
565 m_info->m_rdpos += count;
566 return pplx::task_from_result<size_t>(count);
567 }
568 }
569
570 auto result_tce = pplx::task_completion_event<size_t>();
571 auto callback = new _filestream_callback_read(m_info, result_tce);
572
573 size_t read = _getn_fsb(m_info, callback, ptr, count, sizeof(_CharType));
574
575 if ( read != 0 && read != size_t(-1) )
576 {
577 delete callback;
578 pplx::extensibility::scoped_recursive_lock_t lck(m_info->m_lock);
579 m_info->m_rdpos += read/sizeof(_CharType);
580 return pplx::task_from_result<size_t>(read/sizeof(_CharType));
581 }
582 return pplx::create_task(result_tce);
583 });
584 }

593 size_t _sgetn(_Out_writes_ (count) _CharType *ptr, _In_ size_t count)
594 {
595 m_readOps.wait();
596 if ( m_info->m_atend ) return 0;
597
598 if ( count == 0 || in_avail() == 0 ) return 0;
599
600 pplx::extensibility::scoped_recursive_lock_t lck(m_info->m_lock);
601
602 size_t available = _in_avail_unprot();
603 size_t copy = (count < available) ? count : available;
604
605 auto bufoff = m_info->m_rdpos - m_info->m_bufoff;
606 std::memcpy((void *)ptr, this->m_info->m_buffer+bufoff*sizeof(_CharType), copy*sizeof(_CharType));
607
608 m_info->m_rdpos += copy;
609 m_info->m_atend = (copy < count);
610 return copy;
611 }

645 virtual pos_type seekpos(pos_type pos, std::ios_base::openmode mode)
646 {
647 if ( mode == std::ios_base::in )
648 {
649 m_readOps.wait();
650 return (pos_type)_seekrdpos_fsb(m_info, size_t(pos), sizeof(_CharType));
651 }
652 else if ( (m_info->m_mode & std::ios::ios_base::app) == 0 )
653 {

655 }

668 virtual pos_type seekoff(off_type offset, std::ios_base::seekdir way, std::ios_base::openmode mode)
669 {
670 if ( mode == std::ios_base::in )
671 {
672 m_readOps.wait();
673 switch ( way )
674 {
675 case std::ios_base::beg:
676 return (pos_type)_seekrdpos_fsb(m_info, size_t(offset), sizeof(_CharType));
677 case std::ios_base::cur:
678 return (pos_type)_seekrdpos_fsb(m_info, size_t(m_info->m_rdpos+offset), sizeof(_CharType));
679 case std::ios_base::end:
680 return (pos_type)_seekrdtoend_fsb(m_info, int64_t(offset), sizeof(_CharType));
681 default:
682 // Fail on invalid input (_S_ios_seekdir_end)
683 assert(false);
684 }
685 }
686 else if ( (m_info->m_mode & std::ios::ios_base::app) == 0 )
687 {
688 switch ( way )
689 {
690 case std::ios_base::beg:
691 return (pos_type)_seekwrpos_fsb(m_info, size_t(offset), sizeof(_CharType));
692 case std::ios_base::cur:
693 return (pos_type)_seekwrpos_fsb(m_info, size_t(m_info->m_wrpos+offset), sizeof(_CharType));
694 case std::ios_base::end:
695 return (pos_type)_seekwrpos_fsb(m_info, size_t(-1), sizeof(_CharType));
696 default:
697 // Fail on invalid input (_S_ios_seekdir_end)
698 assert(false);
699 }
700 }
701 return (pos_type)traits::eof();
702 }

715 pplx::task<void> flush_internal()
716 {
717 pplx::task_completion_event<void> result_tce;
718 auto callback = utility::details::make_unique<_filestream_callback_write_b>(m_info, result_tce);

722 return pplx::task_from_exception<void>(std::runtime_error("failure to flush stream"));

724 callback.release();
725 return pplx::create_task(result_tce);
