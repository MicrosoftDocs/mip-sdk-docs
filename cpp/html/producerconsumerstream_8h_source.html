141 virtual _CharType* _alloc(size_t count)
142 {
143 if (!this->can_write())
144 {
145 return nullptr;
146 }
147
148 // We always allocate a new block even if the count could be satisfied by
149 // the current write block. While this does lead to wasted space it allows for
150 // easier book keeping
151
152 _ASSERTE(!m_allocBlock);
153 m_allocBlock = std::make_shared<_block>(count);
154 return m_allocBlock->wbegin();
155 }

161 virtual void _commit(size_t count)
162 {
163 pplx::extensibility::scoped_critical_section_t l(m_lock);
164
165 // The count does not reflect the actual size of the block.
166 // Since we do not allow any more writes to this block it would suffice.
167 // If we ever change the algorithm to reuse blocks then this needs to be revisited.
168
169 _ASSERTE((bool)m_allocBlock);
170 m_allocBlock->update_write_head(count);
171 m_blocks.push_back(m_allocBlock);
172 m_allocBlock = nullptr;
173
174 update_write_head(count);
175 }

190 virtual bool acquire(_Out_ _CharType*& ptr, _Out_ size_t& count)
191 {
192 count = 0;
193 ptr = nullptr;
194
195 if (!this->can_read()) return false;
196
197 pplx::extensibility::scoped_critical_section_t l(m_lock);
198
199 if (m_blocks.empty())
200 {
201 // If the write head has been closed then have reached the end of the
202 // stream (return true), otherwise more data could be written later (return false).
203 return !this->can_write();
204 }
205 else
206 {
207 auto block = m_blocks.front();
208
209 count = block->rd_chars_left();
210 ptr = block->rbegin();
211
212 _ASSERTE(ptr != nullptr);
213 return true;
214 }
215 }
216
223 virtual void release(_Out_writes_opt_ (count) _CharType *ptr, _In_ size_t count)
224 {
225 if (ptr == nullptr) return;
226
227 pplx::extensibility::scoped_critical_section_t l(m_lock);
228 auto block = m_blocks.front();
229
230 _ASSERTE(block->rd_chars_left() >= count);
231 block->m_read += count;
232
233 update_read_head(count);
234 }

337 pplx::task<void> _close_write()
338 {
339 // First indicate that there could be no more writes.
340 // Fulfill outstanding relies on that to flush all the
341 // read requests.
342 this->m_stream_can_write = false;
343
344 {
345 pplx::extensibility::scoped_critical_section_t l(this->m_lock);
346
347 // This runs on the thread that called close.
348 this->fulfill_outstanding();
349 }
350
351 return pplx::task_from_result();
352 }

358 void update_write_head(size_t count)
359 {
360 m_total += count;

368 size_t write(const _CharType *ptr, size_t count)
369 {
370 if (!this->can_write() || (count == 0)) return 0;
371
372 // If no one is going to read, why bother?
373 // Just pretend to be writing!
374 if (!this->can_read()) return count;
375
376 pplx::extensibility::scoped_critical_section_t l(m_lock);
377
378 // Allocate a new block if necessary
379 if ( m_blocks.empty() || m_blocks.back()->wr_chars_left() < count )
380 {
381 SafeInt<size_t> alloc = m_alloc_size.Max(count);
382 m_blocks.push_back(std::make_shared<_block>(alloc));
383 }
384
385 // The block at the back is always the write head
386 auto last = m_blocks.back();
387 auto countWritten = last->write(ptr, count);
388 _ASSERTE(countWritten == count);
389
390 update_write_head(countWritten);
391 return countWritten;
392 }
393
398 void fulfill_outstanding()
399 {
400 while ( !m_requests.empty() )
401 {
402 auto req = m_requests.front();
403
404 // If we cannot satisfy the request then we need
405 // to wait for the producer to write data
406 if (!can_satisfy(req.size())) return;
407
408 // We have enough data to satisfy this request
409 req.complete();
410
411 // Remove it from the request queue
412 m_requests.pop();
413 }

444 // Pointer to the read head
445 _CharType * rbegin()
446 {

451 _CharType * wbegin()

456 // Read up to count characters from the block
457 size_t read(_Out_writes_ (count) _CharType * dest, _In_ size_t count, bool advance = true)
458 {
459 SafeInt<size_t> avail(rd_chars_left());
460 auto countRead = static_cast<size_t>(avail.Min(count));
461
462 _CharType * beg = rbegin();
463 _CharType * end = rbegin() + countRead;
464
465#ifdef _WIN32
466#pragma warning(push)
467#pragma warning(disable: 4996) // Disable warning 4996 (checked_array_iterator needs to be replaced)
468 // Avoid warning C4996: Use checked iterators under SECURE_SCL
469 std::copy(beg, end, stdext::checked_array_iterator<_CharType *>(dest, count));
470#pragma warning(pop)
471#else
472 std::copy(beg, end, dest);
473#endif // _WIN32
474
475 if (advance)
476 {
477 m_read += countRead;
478 }
479
480 return countRead;
481 }
482
483 // Write count characters into the block
484 size_t write(const _CharType * src, size_t count)
485 {
486 SafeInt<size_t> avail(wr_chars_left());
487 auto countWritten = static_cast<size_t>(avail.Min(count));
488
489 const _CharType * srcEnd = src + countWritten;
490
491#ifdef _WIN32
492#pragma warning(push)
493#pragma warning(disable: 4996) // Disable warning 4996 (checked_array_iterator needs to be replaced)
494 // Avoid warning C4996: Use checked iterators under SECURE_SCL
495 std::copy(src, srcEnd, stdext::checked_array_iterator<_CharType *>(wbegin(), static_cast<size_t>(avail)));
496#pragma warning(pop)
497#else
498 std::copy(src, srcEnd, wbegin());
499#endif // _WIN32
500
501 update_write_head(countWritten);
502 return countWritten;
503 }
504
505 void update_write_head(size_t count)
506 {
507 m_pos += count;
508 }
509
510 size_t rd_chars_left() const { return m_pos-m_read; }
511 size_t wr_chars_left() const { return m_size-m_pos; }
512
513 private:
514
515 // Copy is not supported
516 _block(const _block&);
517 _block& operator=(const _block&);
518 };

549 void enqueue_request(_request req)
550 {
551 pplx::extensibility::scoped_critical_section_t l(m_lock);
552
553 if (can_satisfy(req.size()))
554 {
555 // We can immediately fulfill the request.
556 req.complete();
557 }
558 else
559 {
560 // We must wait for data to arrive.
561 m_requests.push(req);
562 }
563 }
564
568 bool can_satisfy(size_t count)
569 {
570 return (m_synced > 0) || (this->in_avail() >= count) || !this->can_write();
571 }
572
578 int_type read_byte(bool advance = true)
579 {
580 _CharType value;
581 auto read_size = this->read(&value, 1, advance);
582 return read_size == 1 ? static_cast<int_type>(value) : traits::eof();
583 }
584
591 size_t read(_Out_writes_ (count) _CharType *ptr, _In_ size_t count, bool advance = true)
592 {
593 _ASSERTE(can_satisfy(count));
594
595 size_t read = 0;
596
597 for (auto iter = begin(m_blocks); iter != std::end(m_blocks); ++iter)
598 {
599 auto block = *iter;
600 auto read_from_block = block->read(ptr + read, count - read, advance);
601
602 read += read_from_block;
603
604 _ASSERTE(count >= read);
605 if (read == count) break;
606 }
607
608 if (advance)
609 {
610 update_read_head(read);
611 }
612
613 return read;
614 }
615
620 void update_read_head(size_t count)
621 {
622 m_total -= count;
623 m_total_read += count;
624
625 if ( m_synced > 0 )
626 m_synced = (m_synced > count) ? (m_synced-count) : 0;
627
628 // The block at the front is always the read head.
629 // Purge empty blocks so that the block at the front reflects the read head
630 while (!m_blocks.empty())
631 {
632 // If front block is not empty - we are done
633 if (m_blocks.front()->rd_chars_left() > 0) break;
634
635 // The block has no more data to be read. Relase the block
636 m_blocks.pop_front();
637 }
638 }

642
643 // Default block size
644 SafeInt<size_t> m_alloc_size;
645
646 // Block used for alloc/commit
647 std::shared_ptr<_block> m_allocBlock;
648
649 // Total available data
650 size_t m_total;
651
652 size_t m_total_read;
653 size_t m_total_written;
654
655 // Keeps track of the number of chars that have been flushed but still
656 // remain to be consumed by a read operation.
657 size_t m_synced;
658
659 // The producer-consumer buffer is intended to be used concurrently by a reader
660 // and a writer, who are not coordinating their accesses to the buffer (coordination
661 // being what the buffer is for in the first place). Thus, we have to protect
662 // against some of the internal data elements against concurrent accesses
663 // and the possibility of inconsistent states. A simple non-recursive lock
664 // should be sufficient for those purposes.
665 pplx::extensibility::critical_section_t m_lock;
666
667 // Memory blocks
668 std::deque<std::shared_ptr<_block>> m_blocks;
669
670 // Queue of requests
671 std::queue<_request> m_requests;
