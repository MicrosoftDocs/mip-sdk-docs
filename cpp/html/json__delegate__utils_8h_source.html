66inline std::shared_ptr<JsonValue> SafeGetMember(const JsonValue& value, const std::string& key) {
67 if (!value.IsObject()) {

173inline OptionalMember SafeGetMemberOptional(const JsonValue& value, const std::string& key, JsonType jsonType) {
174 if (value.HasMember(key)) {
175 auto member = SafeGetMember(value, key);
176 if (CheckJsonType(*member, jsonType)) {
177 return OptionalMember::OfNullable(member);
