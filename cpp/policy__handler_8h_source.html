48/**
49 * @brief This class provides an interface for all policy handler functions on a file.
50 */
51class PolicyHandler {
52public:
53 /**
54 * @brief Get the sensitivity label from existing content.
55 *
56 * @param state Current state of the content.
57 * @return the label currently applied to the content. If not labeled, returns empty.
58 */
59 virtual std::shared_ptr<ContentLabel> GetSensitivityLabel(const ExecutionState& state) = 0;
60
61 /**
62 * @brief Executes the rules in the handler based on the provided state and returns the list of actions
63 * to be executed.
64 *
65 * @param state the current execution state of the content the rules are running on.
66 * @return list of actions that should be applied on the content.
67 */
68 virtual std::vector<std::shared_ptr<Action>> ComputeActions(const ExecutionState& state) = 0;
69
70 /**
71 * @brief Called once the computed actions have been applied, and the data committed to disk.
72 *
73 * @param state the current execution state of the content after the actions have been committed.
74 * @note: This call sends an audit event.
75 */
76 virtual void NotifyCommittedActions(const ExecutionState& state) = 0;
77
78 /**
79 * @brief Checks whether metadata contains label artifacts.
80 *
81 * @param metadata Metadata of file to check
82 * @param mipContext Global MIP context
83 *
84 * @return True if metadata contains label artifacts, else false
85 *
86 * @note This will only detect Microsoft labels. It will not detect 3rd-party labels, even if a tenant is configured
87 * to translate 3rd-party label metadata to Microsoft labels. The primary purpose of this API is to allow an
88 * application to quickly detect labeled content without any HTTP calls, and ths limitation is caused by the
89 * fact that retrieving tenant-specific label mapping would require an HTTP call.
90 */
91 MIP_API static bool __CDECL IsLabeled(
92 const std::vector<std::pair<std::string, std::string>>& metadata,
93 const std::shared_ptr<MipContext>& mipContext);
94
95 /** @cond DOXYGEN_HIDE */
96 virtual ~PolicyHandler() {}
97 /** @endcond */
98};
