1241 storage_type::iterator find_insert_location(const utility::string_t &key)
1242 {
1243 if (m_keep_order)
1244 {
1245 return std::find_if(m_elements.begin(), m_elements.end(),
1246 [&key](const std::pair<utility::string_t, value>& p) {
1247 return p.first == key;
1248 });
1249 }

1251 {

1253 }

1261 [&key](const std::pair<utility::string_t, value>& p) {
1262 return p.first == key;
1263 });
1264 }
1265 else
1266 {
1267 auto iter = std::lower_bound(m_elements.begin(), m_elements.end(), key, compare_with_key);
1268 if (iter != m_elements.end() && key != iter->first)
1269 {
1270 return m_elements.end();
1271 }
1272 return iter;
1273 }
1274 }

1276 storage_type::iterator find_by_key(const utility::string_t& key)
1277 {
1278 auto iter = find_insert_location(key);
1279 if (iter != m_elements.end() && key != iter->first)
1280 {
1281 return m_elements.end();
1282 }
1283 return iter;
1284 }
