70/**
71 * @brief Base class for all errors that will be reported (thrown or returned) from MIP SDK.
72 */
73class Error : public std::exception {
74public:
75 /**
76 * @brief Get the error message.
77 *
78 * @return the error message
79 */
80 char const* what() const noexcept override {
81 return mFormattedMessage.c_str();
82 }
83
84 /**
85 * @brief Clone the error
86 *
87 * @return a clone of the error.
88 */
89 virtual std::shared_ptr<Error> Clone() const = 0;
90
91 /**
92 * @brief Get the error type.
93 *
94 * @return the error type.
95 */
96 virtual ErrorType GetErrorType() const { return mType; }
97
98 /**
99 * @brief Get the error name.
100 *
101 * @return the error name.
102 */
103 const std::string& GetErrorName() const { return mName; }
104
105 /**
106 * @brief Get the error message.
107 *
108 * @return the error message.
109 */
110 const std::string& GetMessage() const {
111 return mFormattedMessage;
112 }
113
114 /**
115 * @brief Set the error message.
116 *
117 * @param msg the error message.
118 */
119 void SetMessage(const std::string& msg) {
120 mMessage = msg;
121 mFormattedMessage = CreateFormattedMessage(msg, mDebugInfo);
122 }
123
124 /**
125 * @brief Add debug info entry
126 *
127 * @param key Debug info key
128 * @param value Debug info value
129 */
130 void AddDebugInfo(const std::string& key, const std::string& value) {
131 if (!key.empty() && !value.empty()) {
132 mDebugInfo[key] = value;
133 mFormattedMessage = mFormattedMessage + ", " + key + "=" + value;
134 }
135 }
136
137 /**
138 * @brief Get debug info
139 *
140 * @return Debug info (keys/values)
141 */
142 const std::map<std::string, std::string>& GetDebugInfo() const { return mDebugInfo; }
143
144 /** @cond DOXYGEN_HIDE */
145 virtual ~Error() {}
146
147protected:
148 explicit Error(
149 const std::string& message,
150 const std::string& name,
151 ErrorType type)
152 : mMessage(message), mName(name), mType(type) {
153 mFormattedMessage = CreateFormattedMessage(message, {{}} /*debugInfo*/);
154 }
155 explicit Error(
156 const std::string& message,
157 const std::map<std::string, std::string> debugInfo,
158 const std::string& name,
159 ErrorType type)
160 : mMessage(message), mDebugInfo(debugInfo), mName(name), mType(type) {
161 mFormattedMessage = CreateFormattedMessage(message, debugInfo);
162 }
163 /** @endcond */
164
165 std::string mMessage;
166 std::map<std::string, std::string> mDebugInfo;
167
168private:
169 std::string CreateFormattedMessage(const std::string& message, const std::map<std::string, std::string> debugInfo) const {
170 auto formattedMessage = message;
171
172 // Remove stray newlines
173 auto isNewlineFn = [](char c) { return c == '\n' || c == '\r'; };
174 formattedMessage.erase(std::remove_if(formattedMessage.begin(), formattedMessage.end(), isNewlineFn), formattedMessage.end());
175
176 // Add debug info entries
177 for (const auto& entry: debugInfo) {
178 if (!entry.first.empty() && !entry.second.empty())
179 formattedMessage = formattedMessage + ", " + entry.first + "=" + entry.second;
180 }
181
182 return formattedMessage;
183 }
184
185 std::string mName;
186 std::string mFormattedMessage;
187 ErrorType mType;
188};

249 * @brief Networking error. Caused by unexpected behavior when making network calls to service endpoints.

251class NetworkError : public Error {
252public:
253 /**
254 * @brief Category of network error
255 */
256 enum class Category {
257 Unknown = 0, /**< Unknown network failure */
258 FailureResponseCode = 1, /**< HTTP response code indicates failure */
259 BadResponse = 2, /**< HTTP response could not be read */
260 UnexpectedResponse = 3, /**< HTTP response completed but contained unexpected data */
261 NoConnection = 4, /**< Failed to establish a connection */
262 Proxy = 5, /**< Proxy failure */
263 SSL = 6, /**< SSL failure */
264 Timeout = 7, /**< Connection timed out */
265 Offline = 8, /**< Operation requires network connectivity */
266 Throttled = 9, /**< HTTP operation failed due to server traffic throttling */
267 Cancelled = 10, /**< HTTP operation has been cancelled by the application */
268 };
269
270 /** @cond DOXYGEN_HIDE */
271 explicit NetworkError(
272 Category category,
273 const std::string& baseUrl,
274 const std::string& requestId,
275 int32_t statusCode,
276 const std::string& message,
277 const std::string& name = "NetworkError")
278 : Error(message, name, ErrorType::NETWORK_ERROR),
279 mCategory(category),
280 mResponseStatusCode(statusCode) {
281 AddDebugInfo("NetworkError.Category", GetCategoryString(category));
282 if (!baseUrl.empty())
283 AddDebugInfo("HttpRequest.BaseUrl", baseUrl);
284 if (!requestId.empty())
285 AddDebugInfo("HttpRequest.Id", requestId);
286 if (statusCode != 0)
287 AddDebugInfo("HttpResponse.StatusCode", std::to_string(static_cast<int>(statusCode)));
288 }
289
290 explicit NetworkError(
291 Category category,
292 int32_t statusCode,
293 const std::string& message,
294 const std::map<std::string, std::string>& debugInfo,
295 const std::string& name = "NetworkError")
296 : Error(message, debugInfo, name, ErrorType::NETWORK_ERROR),
297 mCategory(category),
298 mResponseStatusCode(statusCode) {}
299
300 std::shared_ptr<Error> Clone() const override {
301 return std::make_shared<NetworkError>(mCategory, mResponseStatusCode, mMessage, mDebugInfo);
302 }
303 /** @endcond */
304
305 /**
306 * @brief Gets the category of network failure
307 *
308 * @return Category of network failure
309 */
310 Category GetCategory() const { return mCategory; }
311
312 /**
313 * @brief Gets the HTTP response status code
314 *
315 * @return HTTP response status code, 0 if none
316 */
317 int32_t GetResponseStatusCode() const { return mResponseStatusCode; }
318
319private:
320 Category mCategory;
321 int32_t mResponseStatusCode;
322
323 const std::string& GetCategoryString(Category category) const {
324 static const std::string kUnrecognized = "UNRECOGNIZED";
325 static const std::map<Category, std::string> kCategories = {
326 { Category::Unknown, "Unknown" },
327 { Category::FailureResponseCode, "FailureResponseCode" },
328 { Category::BadResponse, "BadResponse" },
329 { Category::UnexpectedResponse, "UnexpectedResponse" },
330 { Category::NoConnection, "NoConnection" },
331 { Category::Proxy, "Proxy" },
332 { Category::SSL, "SSL" },
333 { Category::Timeout, "Timeout" },
334 { Category::Offline, "Offline" },
335 { Category::Throttled, "Throttled" },
336 { Category::Cancelled, "Cancelled" },
337 };
338 return kCategories.count(category) ? kCategories.at(category) : kUnrecognized;
339 }
