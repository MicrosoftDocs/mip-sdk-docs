126 // Base class for all reference counted objects
127 class _RefCounter
128 {
129 public:
130
131 virtual ~_RefCounter()
132 {
133 _ASSERTE(_M_refCount == 0);
134 }
135
136 // Acquires a reference
137 // Returns the new reference count.
138 long _Reference()
139 {
140 long _Refcount = atomic_increment(_M_refCount);
141
142 // 0 - 1 transition is illegal
143 _ASSERTE(_Refcount > 1);
144 return _Refcount;
145 }
146
147 // Releases the reference
148 // Returns the new reference count
149 long _Release()
150 {
151 long _Refcount = atomic_decrement(_M_refCount);
152 _ASSERTE(_Refcount >= 0);
153
154 if (_Refcount == 0)
155 {
156 _Destroy();
157 }
158
159 return _Refcount;
160 }
161
162 protected:
163
164 // Allow derived classes to provide their own deleter
165 virtual void _Destroy()
166 {
167 delete this;
168 }
169
170 // Only allow instantiation through derived class
171 _RefCounter(long _InitialCount = 1) : _M_refCount(_InitialCount)
172 {
173 _ASSERTE(_M_refCount > 0);
174 }
175
176 // Reference count
177 atomic_long _M_refCount;
178 };

186 static const long _STATE_CLEAR = 0;
187 static const long _STATE_DEFER_DELETE = 1;
188 static const long _STATE_SYNCHRONIZE = 2;
189 static const long _STATE_CALLED = 3;
190
191 public:
192
193 _CancellationTokenRegistration(long _InitialRefs = 1) :
194 _RefCounter(_InitialRefs),
195 _M_state(_STATE_CALLED),
196 _M_pTokenState(NULL)
197 {
198 }
199
200 _CancellationTokenState *_GetToken() const
201 {
202 return _M_pTokenState;
203 }
204
205 protected:
206
207 virtual ~_CancellationTokenRegistration()
208 {
209 _ASSERTE(_M_state != _STATE_CLEAR);
210 }
211
212 virtual void _Exec() = 0;
213
214 private:
215
216 friend class _CancellationTokenState;
217
218 void _Invoke()
219 {
220 long tid = ::pplx::details::platform::GetCurrentThreadId();
221 _ASSERTE((tid & 0x3) == 0); // If this ever fires, we need a different encoding for this.
222
223 long result = atomic_compare_exchange(_M_state, tid, _STATE_CLEAR);
224
225 if (result == _STATE_CLEAR)
226 {
227 _Exec();
228
229 result = atomic_compare_exchange(_M_state, _STATE_CALLED, tid);
230
231 if (result == _STATE_SYNCHRONIZE)
232 {
233 _M_pSyncBlock->set();
234 }
235 }
236 _Release();
237 }
238
239 atomic_long _M_state;
240 extensibility::event_t *_M_pSyncBlock;
241 _CancellationTokenState *_M_pTokenState;

292 protected:
293 class TokenRegistrationContainer
294 {
295 private:
296 typedef struct _Node {
297 _CancellationTokenRegistration* _M_token;
298 _Node *_M_next;
299 } Node;
300
301 public:
302 TokenRegistrationContainer() : _M_begin(nullptr), _M_last(nullptr)
303 {
304 }
305
306 ~TokenRegistrationContainer()
307 {
308#if defined(_MSC_VER)
309#pragma warning(push)
310#pragma warning(disable: 6001)
311#endif
312 auto node = _M_begin;
313 while (node != nullptr)
314 {
315 Node* tmp = node;
316 node = node->_M_next;
317 ::free(tmp);
318 }
319#if defined(_MSC_VER)
320#pragma warning(pop)
321#endif
322 }
323
324 void swap(TokenRegistrationContainer& list)
325 {
326 std::swap(list._M_begin, _M_begin);
327 std::swap(list._M_last, _M_last);
328 }
329
330 bool empty()
331 {
332 return _M_begin == nullptr;
333 }
334
335 template<typename T>
336 void for_each(T lambda)
337 {
338 Node* node = _M_begin;
339
340 while (node != nullptr)
341 {
342 lambda(node->_M_token);
343 node = node->_M_next;
344 }
345 }
346
347 void push_back(_CancellationTokenRegistration* token)
348 {
349 Node* node = reinterpret_cast<Node*>(::malloc(sizeof(Node)));
350 if (node == nullptr)
351 {
352 throw ::std::bad_alloc();
353 }
354
355 node->_M_token = token;
356 node->_M_next = nullptr;
357
358 if (_M_begin == nullptr)
359 {
360 _M_begin = node;
361 }
362 else
363 {
364 _M_last->_M_next = node;
365 }
366
367 _M_last = node;
368 }
369
370 void remove(_CancellationTokenRegistration* token)
371 {
372 Node* node = _M_begin;
373 Node* prev = nullptr;
374
375 while (node != nullptr)
376 {
377 if (node->_M_token == token) {
378 if (prev == nullptr)
379 {
380 _M_begin = node->_M_next;
381 }
382 else
383 {
384 prev->_M_next = node->_M_next;
385 }
386
387 if (node->_M_next == nullptr)
388 {
389 _M_last = prev;
390 }
391
392 ::free(node);
393 break;
394 }
395
396 prev = node;
397 node = node->_M_next;
398 }
399 }
400
401 private:
402 Node *_M_begin;
403 Node *_M_last;
404 };

423 _CancellationTokenState() :
424 _M_stateFlag(0)
425 {
426 }
427
428 ~_CancellationTokenState()
429 {
430 TokenRegistrationContainer rundownList;
431 {
432 extensibility::scoped_critical_section_t _Lock(_M_listLock);
433 _M_registrations.swap(rundownList);
434 }
435
436 rundownList.for_each([](_CancellationTokenRegistration * pRegistration)
437 {
438 pRegistration->_M_state = _CancellationTokenRegistration::_STATE_SYNCHRONIZE;
439 pRegistration->_Release();
440 });
441 }
442
443 bool _IsCanceled() const
444 {
445 return (_M_stateFlag != 0);
446 }
447
448 void _Cancel()
449 {
450 if (atomic_compare_exchange(_M_stateFlag, 1l, 0l) == 0)
451 {
452 TokenRegistrationContainer rundownList;
453 {
454 extensibility::scoped_critical_section_t _Lock(_M_listLock);
455 _M_registrations.swap(rundownList);
456 }
457
458 rundownList.for_each([](_CancellationTokenRegistration * pRegistration)
459 {
460 pRegistration->_Invoke();
461 });
462
463 _M_stateFlag = 2;
464 _M_cancelComplete.set();
465 }
466 }
467
468 _CancellationTokenRegistration *_RegisterCallback(TaskProc_t _PCallback, _In_ void *_PData, int _InitialRefs = 1)
469 {
470 _CancellationTokenRegistration *pRegistration = new CancellationTokenRegistration_TaskProc(_PCallback, _PData, _InitialRefs);
471 _RegisterCallback(pRegistration);
472 return pRegistration;
473 }
474
475 void _RegisterCallback(_In_ _CancellationTokenRegistration *_PRegistration)
476 {
477 _PRegistration->_M_state = _CancellationTokenRegistration::_STATE_CLEAR;
478 _PRegistration->_Reference();
479 _PRegistration->_M_pTokenState = this;
480
481 bool invoke = true;
482
483 if (!_IsCanceled())
484 {
485 extensibility::scoped_critical_section_t _Lock(_M_listLock);
486
487 if (!_IsCanceled())
488 {
489 invoke = false;
490 _M_registrations.push_back(_PRegistration);
491 }
492 }
493
494 if (invoke)
495 {
496 _PRegistration->_Invoke();
497 }
498 }
499
500 void _DeregisterCallback(_In_ _CancellationTokenRegistration *_PRegistration)
501 {
502 bool synchronize = false;
503
504 {
505 extensibility::scoped_critical_section_t _Lock(_M_listLock);
506
507 //
508 // If a cancellation has occurred, the registration list is guaranteed to be empty if we've observed it under the auspices of the
509 // lock. In this case, we must synchronize with the cancelling thread to guarantee that the cancellation is finished by the time
510 // we return from this method.
511 //
512 if (!_M_registrations.empty())
513 {
514 _M_registrations.remove(_PRegistration);
515 _PRegistration->_M_state = _CancellationTokenRegistration::_STATE_SYNCHRONIZE;
516 _PRegistration->_Release();
517 }
518 else
519 {
520 synchronize = true;
521 }
522 }
523
524 //
525 // If the list is empty, we are in one of several situations:
526 //
527 // - The callback has already been made --> do nothing
528 // - The callback is about to be made --> flag it so it doesn't happen and return
529 // - The callback is in progress elsewhere --> synchronize with it
530 // - The callback is in progress on this thread --> do nothing
531 //
532 if (synchronize)
533 {
534 long result = atomic_compare_exchange(
535 _PRegistration->_M_state,
536 _CancellationTokenRegistration::_STATE_DEFER_DELETE,
537 _CancellationTokenRegistration::_STATE_CLEAR
538 );
539
540 switch(result)
541 {
542 case _CancellationTokenRegistration::_STATE_CLEAR:
543 case _CancellationTokenRegistration::_STATE_CALLED:
544 break;
545 case _CancellationTokenRegistration::_STATE_DEFER_DELETE:
546 case _CancellationTokenRegistration::_STATE_SYNCHRONIZE:
547 _ASSERTE(false);
548 break;
549 default:
550 {
551 long tid = result;
552 if (tid == ::pplx::details::platform::GetCurrentThreadId())
553 {
554 //
555 // It is entirely legal for a caller to Deregister during a callback instead of having to provide their own synchronization
556 // mechanism between the two. In this case, we do *NOT* need to explicitly synchronize with the callback as doing so would
557 // deadlock. If the call happens during, skip any extra synchronization.
558 //
559 break;
560 }
561
562 extensibility::event_t ev;
563 _PRegistration->_M_pSyncBlock = &ev;
564
565 long result_1 = atomic_exchange(_PRegistration->_M_state, _CancellationTokenRegistration::_STATE_SYNCHRONIZE);
566
567 if (result_1 != _CancellationTokenRegistration::_STATE_CALLED)
568 {
569 _PRegistration->_M_pSyncBlock->wait(::pplx::extensibility::event_t::timeout_infinite);
570 }
571
572 break;
573 }
574 }
575 }
576 }
577
578 private:
579
580 // The flag for the token state (whether it is canceled or not)
581 atomic_long _M_stateFlag;
582
583 // Notification of completion of cancellation of this token.
584 extensibility::event_t _M_cancelComplete; // Hmm.. where do we wait for it??
585
586 // Lock to protect the registrations list
587 extensibility::critical_section_t _M_listLock;
588
589 // The protected list of registrations
590 TokenRegistrationContainer _M_registrations;
591 };
